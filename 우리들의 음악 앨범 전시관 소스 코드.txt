    function escapeHtml(value) {
        return String(value ?? '')
            .replaceAll('&', '&amp;')
            .replaceAll('<', '&lt;')
            .replaceAll('>', '&gt;')
            .replaceAll('"', '&quot;')
            .replaceAll("'", '&#039;');
    }

    function safeExternalUrl(value) {
        try {
            const url = new URL(String(value));
            return ['https:', 'http:'].includes(url.protocol) ? url.href : '#';
        } catch {
            return '#';
        }
    }

    function safeImageUrl(value) {
        const url = String(value || '');
        if (/^data:image\/(jpeg|jpg|png|webp);base64,/i.test(url)) return url;
        return safeExternalUrl(url);
    }

    // 다른 기기에서도 표지가 보이도록 이미지를 작게 압축해 Firestore에 저장한다.
    function compressCoverImage(file) {
        return new Promise((resolve, reject) => {
            const reader = new FileReader();

            reader.onerror = () => reject(new Error('이미지 파일을 읽을 수 없습니다.'));
            reader.onload = () => {
                const image = new Image();
                image.onerror = () => reject(new Error('지원하지 않는 이미지 형식입니다.'));
                image.onload = () => {
                    const sourceSize = Math.min(image.width, image.height);
                    const sourceX = (image.width - sourceSize) / 2;
                    const sourceY = (image.height - sourceSize) / 2;
                    let outputSize = 600;
                    let quality = 0.78;

                    const makeDataUrl = () => {
                        const canvas = document.createElement('canvas');
                        canvas.width = outputSize;
                        canvas.height = outputSize;
                        const context = canvas.getContext('2d');
                        context.fillStyle = '#191414';
                        context.fillRect(0, 0, outputSize, outputSize);
                        context.drawImage(
                            image,
                            sourceX,
                            sourceY,
                            sourceSize,
                            sourceSize,
                            0,
                            0,
                            outputSize,
                            outputSize
                        );
                        return canvas.toDataURL('image/jpeg', quality);
                    };

                    let dataUrl = makeDataUrl();
                    if (dataUrl.length > 700000) {
                        outputSize = 420;
                        quality = 0.65;
                        dataUrl = makeDataUrl();
                    }

                    if (dataUrl.length > 850000) {
                        reject(new Error('이미지 용량이 너무 큽니다. 다른 이미지를 선택해주세요.'));
                        return;
                    }
                    resolve(dataUrl);
                };
                image.src = reader.result;
            };
            reader.readAsDataURL(file);
        });
    }

    function renderGallery() {
        const gallery = document.getElementById('album-gallery');
        const resultCount = document.getElementById('search-result-count');
        gallery.innerHTML = '';

        const query = currentSearchQuery.trim().toLowerCase();

        const filteredAlbums = albumsData.filter((album) => {
            const matchesClass = currentFilter === 'all' || album.classNum === currentFilter;
            if (!matchesClass) return false;
            if (!query) return true;

            // 앨범명, 학생/팀 이름, 반, 앨범 소개, 제작진, 수록곡 제목/설명을 한 번에 검색한다.
            const credits = album.credits || {};
            const tracks = Array.isArray(album.tracks) ? album.tracks : [];
            const searchableText = [
                album.title,
                album.artist,
                album.classNum,
                album.description,
                credits.lyricist,
                credits.composer,
                credits.producer,
                ...tracks.flatMap(track => [track?.title, track?.desc])
            ]
                .filter(Boolean)
                .join(' ')
                .toLowerCase();

            return searchableText.includes(query);
        });

        if (resultCount) {
            if (query) {
                resultCount.textContent = `검색 결과 ${filteredAlbums.length}개`;
            } else if (currentFilter !== 'all') {
                resultCount.textContent = `${currentFilter} 앨범 ${filteredAlbums.length}개`;
            } else {
                resultCount.textContent = '';
            }
        }

        if (filteredAlbums.length === 0) {
            const hasSearch = Boolean(query);
            gallery.innerHTML = `
                <div class="col-span-full text-center py-20 text-gray-500">
                    <i class="fas ${hasSearch ? 'fa-search' : 'fa-inbox'} text-4xl mb-4"></i>
                    <p>${hasSearch
                        ? '검색 결과가 없습니다.<br>다른 검색어를 입력해보세요.'
                        : '아직 등록된 앨범이 없습니다.<br>첫 번째 앨범을 등록해보세요!'
                    }</p>
                </div>`;
            return;
        }

        filteredAlbums.forEach(album => {
            const card = document.createElement('div');
            card.className = 'album-card bg-gray-900 rounded-lg md:rounded-xl p-1 md:p-4 cursor-pointer group relative';
            card.onclick = () => showAlbumDetail(album.id);
            const coverUrl = safeImageUrl(album.coverUrl) || 'https://placehold.co/600x600/1db954/ffffff?text=No+Cover';

            // 이미지 에러 처리 추가
            card.innerHTML = `

    function renderRatingPanel(ratings = currentRatings) {
        const summary = document.getElementById('rating-summary');
        const badge = document.getElementById('rating-average-badge');
        const myStatus = document.getElementById('my-rating-status');
        const buttons = document.querySelectorAll('[data-rating-score]');

        if (!summary || !badge || !myStatus) return;

        const validRatings = (Array.isArray(ratings) ? ratings : []).filter(rating => {
            const score = Number(rating.score);
            return Number.isInteger(score) && score >= 1 && score <= 5;
        });

        const count = validRatings.length;
        const average = count
            ? validRatings.reduce((sum, rating) => sum + Number(rating.score), 0) / count
            : 0;

        const myRating = validRatings.find(rating =>
            rating.userId === currentUserId || rating.id === currentUserId
        );
        const myScore = myRating ? Number(myRating.score) : 0;

        if (count > 0) {
            summary.textContent = `현재 ${count}명이 평가했습니다.`;
            badge.innerHTML = `<i class="fas fa-star text-yellow-400 mr-1"></i>${average.toFixed(1)} / 5 · ${count}명`;
            badge.classList.remove('text-gray-300');
            badge.classList.add('text-white');
        } else {
            summary.textContent = '아직 등록된 별점이 없습니다.';
            badge.textContent = '아직 평가 없음';
            badge.classList.add('text-gray-300');
            badge.classList.remove('text-white');
        }

        buttons.forEach(button => {
            const score = Number(button.dataset.ratingScore);
            button.classList.toggle('text-yellow-400', score <= myScore);
            button.classList.toggle('text-gray-600', score > myScore);
            button.setAttribute('aria-pressed', score === myScore ? 'true' : 'false');
        });

        myStatus.textContent = myScore
            ? `내 별점: ${myScore}점 · 다른 별을 누르면 내 점수만 수정됩니다.`
            : '별을 눌러 1~5점으로 평가해주세요.';
    }

    async function startRatingListener(albumId) {
        stopRatingListener();
        renderRatingPanel([]);
        setRatingButtonsDisabled(true);

        const summary = document.getElementById('rating-summary');
        const myStatus = document.getElementById('my-rating-status');
        if (summary) summary.textContent = '별점을 불러오는 중...';
        if (myStatus) myStatus.textContent = 'Firebase 연결을 확인하는 중입니다.';

        try {
            const service = await window.firebaseAlbumServicePromise;
            currentUserId = service.currentUserId;
            setRatingButtonsDisabled(false);

            ratingUnsubscribe = service.listenRatings(
                albumId,
                (ratings) => {
                    if (currentDetailAlbumId !== albumId) return;
                    currentRatings = ratings;
                    renderRatingPanel(ratings);
                },
                (error) => {
                    console.error('별점 실시간 불러오기 실패:', error);
                    if (summary) summary.textContent = '별점을 불러오지 못했습니다.';
                    if (myStatus) myStatus.textContent = 'Firestore 별점 권한 규칙을 확인해주세요.';
                    setRatingButtonsDisabled(true);
                }
            );
        } catch (error) {
            console.error('별점 연결 실패:', error);
            if (summary) summary.textContent = '별점 연결에 실패했습니다.';
            if (myStatus) myStatus.textContent = '잠시 후 새로고침해주세요.';
            setRatingButtonsDisabled(true);
        }
    }

    async function submitRating(score) {
        if (!currentDetailAlbumId || ratingSubmitting) return;
        if (!Number.isInteger(score) || score < 1 || score > 5) return;

        ratingSubmitting = true;
        setRatingButtonsDisabled(true);
        const myStatus = document.getElementById('my-rating-status');
        if (myStatus) myStatus.textContent = `${score}점을 저장하는 중...`;

        try {
            const service = await window.firebaseAlbumServicePromise;
            currentUserId = service.currentUserId;
            await service.saveRating(currentDetailAlbumId, score);
            // 저장 직후 onSnapshot이 다시 실행되어 평균과 내 별점이 자동 갱신됩니다.
        } catch (error) {
            console.error('별점 저장 실패:', error);
            showAlert(
                '별점 저장 실패',
                `${error.message || '별점을 저장하지 못했습니다.'}<br>Firestore 별점 규칙을 확인해주세요.`,
                'error'
            );
            renderRatingPanel(currentRatings);
        } finally {
            ratingSubmitting = false;
            if (currentDetailAlbumId) setRatingButtonsDisabled(false);
        }
    }

    function getPlaylistItems() {
        try {
            const saved = JSON.parse(localStorage.getItem(PLAYLIST_STORAGE_KEY) || '[]');
            return Array.isArray(saved) ? saved : [];
        } catch (error) {
            console.warn('Playlist 불러오기 실패:', error);
            return [];
        }
    }

    function savePlaylistItems(items) {
        localStorage.setItem(PLAYLIST_STORAGE_KEY, JSON.stringify(items));
        updatePlaylistBadge();
    }

    function makePlaylistItem(album, track, trackIndex) {
        return {
            id: `${String(album.id)}::${trackIndex}`,
            albumId: String(album.id),
            trackIndex,
            trackTitle: String(track?.title || `Track ${trackIndex + 1}`),
            trackLink: String(track?.link || ''),
            trackDesc: String(track?.desc || ''),
            albumTitle: String(album.title || ''),
            artist: String(album.artist || ''),
            classNum: String(album.classNum || ''),
            addedAt: Date.now()
        };
    }

    function isPlaylistItemSaved(itemId) {
        return getPlaylistItems().some(item => item.id === itemId);
    }

    function updatePlaylistBadge() {
        const badge = document.getElementById('playlist-count-badge');
        if (!badge) return;
        const count = getPlaylistItems().length;
        badge.textContent = String(count);
        badge.classList.toggle('hidden', count === 0);
        badge.classList.toggle('inline-flex', count > 0);
    }

    function applyPlaylistButtonState(button, isSaved) {
        if (!button) return;
        button.innerHTML = isSaved
            ? '<i class="fas fa-check text-xs sm:text-sm"></i>'
            : '<i class="fas fa-plus text-xs sm:text-sm"></i>';
        button.setAttribute('aria-label', isSaved ? 'Playlist에 추가됨' : 'Playlist에 추가');
        button.setAttribute('title', isSaved ? 'Playlist에 추가됨' : 'Playlist에 추가');
        button.classList.toggle('bg-spotify-base', isSaved);
        button.classList.toggle('hover:bg-spotify-dark', isSaved);
        button.classList.toggle('text-black', isSaved);
        button.classList.toggle('bg-gray-800', !isSaved);
        button.classList.toggle('hover:bg-gray-700', !isSaved);
        button.classList.toggle('text-white', !isSaved);
    }

    function togglePlaylistTrack(album, track, trackIndex, button) {
        const playlistItem = makePlaylistItem(album, track, trackIndex);
        const items = getPlaylistItems();
        const existingIndex = items.findIndex(item => item.id === playlistItem.id);
        let isSaved;

        if (existingIndex >= 0) {
            items.splice(existingIndex, 1);
            isSaved = false;
        } else {
            items.unshift(playlistItem);
            isSaved = true;
        }

        savePlaylistItems(items);
        applyPlaylistButtonState(button, isSaved);
    }

    function removePlaylistItem(itemId) {
        const items = getPlaylistItems().filter(item => item.id !== itemId);
        savePlaylistItems(items);
        renderPlaylist();
    }

    function isOwnedAlbum(album) {

    function handleTrackTitleSecretClick(albumId, trackIndex, titleElement) {
        const now = Date.now();
        let clickTimes = [];

        try {
            clickTimes = JSON.parse(titleElement.dataset.secretClickTimes || '[]');
            if (!Array.isArray(clickTimes)) clickTimes = [];
        } catch {
            clickTimes = [];
        }

        // 오래된 클릭은 버리고 최근 6초 안의 클릭만 본다.
        clickTimes = clickTimes.filter(time => now - Number(time) <= 6000);
        clickTimes.push(now);

        // 최근 6번 클릭이 '더블클릭 3회' 형태인지 확인한다.
        // 각 더블클릭의 두 클릭 간격은 650ms 이하로 넉넉하게 허용한다.
        if (clickTimes.length >= 6) {
            const lastSix = clickTimes.slice(-6);
            const threeDoubleClicks =
                (lastSix[1] - lastSix[0] <= 650) &&
                (lastSix[3] - lastSix[2] <= 650) &&
                (lastSix[5] - lastSix[4] <= 650);

            if (threeDoubleClicks) {
                titleElement.dataset.secretClickTimes = '[]';
                toggleTrackTitleFromDetail(albumId, trackIndex);
                return;
            }
        }

        titleElement.dataset.secretClickTimes = JSON.stringify(clickTimes.slice(-6));
    }

    async function toggleTrackTitleFromDetail(albumId, trackIndex) {
        const albumIndex = albumsData.findIndex(album => String(album.id) === String(albumId));
        if (albumIndex < 0) return;

        const album = albumsData[albumIndex];
        const tracks = Array.isArray(album.tracks) ? album.tracks : [];
        if (!tracks[trackIndex]) return;

        const wasTitle = Boolean(tracks[trackIndex].isTitle);
        const updatedTracks = tracks.map((track, index) => ({
            ...track,
            isTitle: index === trackIndex ? !wasTitle : false
        }));
        const updatedAlbum = { ...album, tracks: updatedTracks };

        // 세 번째 더블클릭이 인식됐다는 것을 즉시 보여준다.
        albumsData[albumIndex] = updatedAlbum;
        updateDetailTitleBadges(updatedAlbum);

        try {
            const service = await window.firebaseAlbumServicePromise;
            await service.updateAlbum(updatedAlbum);
        } catch (error) {
            // 저장 실패 시 화면과 메모리 상태를 원래대로 되돌린다.
            albumsData[albumIndex] = album;
            updateDetailTitleBadges(album);
            console.error('타이틀곡 저장 실패:', error);
            showAlert(
                '타이틀곡 저장 실패',
                `${error.message || '타이틀곡 정보를 저장하지 못했습니다.'}<br>Firestore 수정 권한을 확인해주세요.`,
                'error'
            );
        }
    }

    async function deleteAlbum(id) {
        const album = albumsData.find(a => a.id === id);
        if (!album || !isOwnedAlbum(album)) {
            showAlert('삭제 불가', '본인이 등록한 앨범만 삭제할 수 있습니다.', 'error');
            return;
        }

        const confirmed = window.confirm(`“${album.title}” 앨범을 정말 삭제할까요?\n삭제한 데이터는 되돌릴 수 없습니다.`);
        if (!confirmed) return;

        try {
            const service = await window.firebaseAlbumServicePromise;
            await service.deleteAlbum(id);
            closeAlbumDetail();
            showAlert('삭제 완료', '앨범이 삭제되었습니다.', 'success');
        } catch (error) {
            console.error('앨범 삭제 실패:', error);
            showAlert('삭제 실패', `${error.message || '앨범을 삭제하지 못했습니다.'}<br>Firestore 규칙을 확인해주세요.`, 'error');
        }
    }

    async function handleRegistration(e) {
        e.preventDefault();

        const isEditing = Boolean(editingAlbumId);
        const targetAlbumId = editingAlbumId;
        const submitButton = e.submitter || e.target.querySelector('button[type="submit"]');
        if (submitButton) {
            submitButton.disabled = true;
            submitButton.innerHTML = `<i class="fas fa-circle-notch fa-spin mr-2"></i>${isEditing ? '수정 중...' : '등록 중...'}`;
        }

        try {
            const service = await window.firebaseAlbumServicePromise;
            const coverFile = document.getElementById('reg-cover').files[0];
            let coverUrl = editingOriginalCoverUrl;

            if (coverFile) {
                coverUrl = await compressCoverImage(coverFile);
            } else if (!isEditing) {
                throw new Error('앨범 자켓 이미지를 선택해주세요.');
            }

            const trackItems = document.querySelectorAll('#tracks-container .track-input-item');
            const tracks = [];

            trackItems.forEach((item, index) => {
                const titleInput = item.querySelector('.track-title');
                const linkInput = item.querySelector('.track-link');
                const descInput = item.querySelector('.track-desc');
                const title = titleInput ? titleInput.value.trim() : '';
                const link = linkInput ? linkInput.value.trim() : '';
                const desc = descInput ? descInput.value.trim() : '';
                const isTitle = item.dataset.isTitle === 'true';

                if (index === 0 || title || link) {
                    tracks.push({
                        title: title || `Track ${index + 1}`,
                        link: link || '#',
                        desc: desc || '',
                        isTitle
                    });
                }
            });

            const albumData = {
                id: isEditing
                    ? targetAlbumId
                    : ((window.crypto && crypto.randomUUID)
                        ? crypto.randomUUID()
                        : `album_${Date.now()}_${Math.random().toString(36).slice(2)}`),
                classNum: document.getElementById('reg-class').value,
                artist: document.getElementById('reg-artist').value.trim(),
                title: document.getElementById('reg-title').value.trim(),
                description: document.getElementById('reg-description').value.trim(),
                coverUrl: coverUrl,
                credits: {
                    lyricist: document.getElementById('reg-lyricist').value.trim(),
                    composer: document.getElementById('reg-composer').value.trim(),
                    producer: document.getElementById('reg-producer').value.trim()
                },
                tracks: tracks
            };

            if (isEditing) {
                await service.updateAlbum(albumData);
                closeRegistrationModal();
                showAlert('수정 완료!', '앨범 정보가 수정되었습니다.', 'success');
            } else {
                await service.saveAlbum(albumData);
                closeRegistrationModal();
                showAlert('등록 완료!', '앨범이 공동 전시관에 저장되었습니다.<br>다른 기기에도 바로 표시됩니다.', 'success');
            }
        } catch (error) {
            console.error(isEditing ? '앨범 수정 실패:' : '앨범 등록 실패:', error);
            showAlert(
                isEditing ? '수정 실패' : '등록 실패',
                `${error.message || '데이터베이스에 저장하지 못했습니다.'}<br>잠시 후 다시 시도해주세요.`,
                'error'
            );
        } finally {
            if (submitButton) {
                submitButton.disabled = false;
                submitButton.textContent = isEditing ? '수정 저장' : '앨범 등록하기';
            }
        }
    }

    function showAlert(title, message, type = 'info') {

        const titleEl = document.getElementById('alert-title');
        const messageEl = document.getElementById('alert-message');

        titleEl.textContent = title;
        messageEl.innerHTML = message;

        // 아이콘 색상 설정
        iconEl.className = 'fas mb-4 text-5xl ';

        setTimeout(() => {
            content.classList.remove('scale-95');

    }

    window.firebaseAlbumServicePromise = (async () => {
        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);
        const credential = await signInAnonymously(auth);
        const albumsCollection = collection(
            db,
            'artifacts',
            'music-album-gallery',
            'public',
            'data',
            'albums'
        );

        return {
            currentUserId: credential.user.uid,

            saveAlbum: async (album) => {
                await setDoc(doc(albumsCollection, String(album.id)), {
                    ...album,
                    createdAt: serverTimestamp(),
                    createdBy: credential.user.uid
                });
            },

            updateAlbum: async (album) => {
                await setDoc(
                    doc(albumsCollection, String(album.id)),
                    {
                        ...album,
                        updatedAt: serverTimestamp()
                    },
                    { merge: true }
                );
            },

            deleteAlbum: async (albumId) => {
                await deleteDoc(doc(albumsCollection, String(albumId)));
            },

            saveRating: async (albumId, score) => {
                const numericScore = Number(score);
                if (!Number.isInteger(numericScore) || numericScore < 1 || numericScore > 5) {
                    throw new Error('별점은 1점부터 5점까지만 입력할 수 있습니다.');
                }

                const ratingsCollection = collection(
                    doc(albumsCollection, String(albumId)),
                    'ratings'
                );
                const ratingDoc = doc(ratingsCollection, credential.user.uid);

                // 문서 ID를 익명 로그인 UID로 사용하므로 한 사용자당 한 개의 별점만 존재합니다.
                // 다시 평가하면 같은 문서를 덮어써서 중복 집계되지 않습니다.
                await setDoc(
                    ratingDoc,
                    {
                        albumId: String(albumId),
                        userId: credential.user.uid,
                        score: numericScore,
                        updatedAt: serverTimestamp()
                    },
                    { merge: true }
                );
            },

            listenRatings: (albumId, onData, onError) => {
                const ratingsCollection = collection(
                    doc(albumsCollection, String(albumId)),
                    'ratings'
                );

                return onSnapshot(
                    ratingsCollection,
                    (snapshot) => {
                        const ratings = snapshot.docs.map((ratingDoc) => ({
                            id: ratingDoc.id,
                            ...ratingDoc.data()
                        }));
                        onData(ratings);
                    },
                    onError
                );
            },

            listenAlbums: (onData, onError) => {
                return onSnapshot(
                    albumsCollection,
                    (snapshot) => {
                        const albums = snapshot.docs
                            .map((albumDoc) => {
                                const data = albumDoc.data();
                                return {
                                    ...data,
                                    id: albumDoc.id,
                                    createdAtMillis: data.createdAt?.toMillis?.() || 0
                                };
                            })
                            .sort((a, b) => b.createdAtMillis - a.createdAtMillis);
                        onData(albums);
                    },
                    onError
                );
            }
        };
    })();
